cmake_minimum_required(VERSION 3.20)
project(HydraPulseField VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(HPF_ENABLE_SANITIZERS "Address + undefined behavior sanitizers" OFF)
option(HPF_UBSAN_ONLY "Undefined behavior sanitizer only" OFF)
option(HPF_ADAPTER_CONTRACT_CHECK "Compile adapters against an explicit host stub, NOT ARM" ON)
add_library(hydrapulse_core INTERFACE)
target_include_directories(hydrapulse_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(hydrapulse_core INTERFACE /W4 /permissive- /fp:precise)
else()
  target_compile_options(hydrapulse_core INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow
    -Werror -fno-fast-math -ffp-contract=off)
endif()
if(HPF_ENABLE_SANITIZERS OR HPF_UBSAN_ONLY)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Requested sanitizers require GCC or Clang")
  endif()
  if(HPF_ENABLE_SANITIZERS)
    set(HPF_SANITIZERS address,undefined)
  else()
    set(HPF_SANITIZERS undefined)
  endif()
  target_compile_options(hydrapulse_core INTERFACE -fsanitize=${HPF_SANITIZERS} -fno-omit-frame-pointer)
  target_link_options(hydrapulse_core INTERFACE -fsanitize=${HPF_SANITIZERS} -fno-omit-frame-pointer)
endif()
add_library(hydrapulse_engine STATIC src/dsp/Voices.cpp src/app/Engine.cpp src/app/Controller.cpp)
target_link_libraries(hydrapulse_engine PUBLIC hydrapulse_core)
if(HPF_ADAPTER_CONTRACT_CHECK)
  foreach(app IN ITEMS FieldTruth HydraPulse)
    add_library(contract_${app} OBJECT firmware/${app}.cpp)
    target_include_directories(contract_${app} BEFORE PRIVATE tests/adapter_stub)
    target_link_libraries(contract_${app} PRIVATE hydrapulse_core)
  endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_subdirectory(tests/native)
endif()
add_executable(render_scenarios tools/render_scenarios.cpp)
target_link_libraries(render_scenarios PRIVATE hydrapulse_engine)
